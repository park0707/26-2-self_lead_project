"""
    canonical_seq: (64, 33, 3) 64프레임 정규 관절 시계열
    """
# 관절 추출: Hip(23/24), Knee(25/26), Ankle(27/28), Foot(31/32)
    hip = 0.5 * (canonical_seq[:, 23, :] + canonical_seq[:, 24, :])
    knee = 0.5 * (canonical_seq[:, 25, :] + canonical_seq[:, 26, :])
    ankle = 0.5 * (canonical_seq[:, 27, :] + canonical_seq[:, 28, :])
    toe = 0.5 * (canonical_seq[:, 31, :] + canonical_seq[:, 32, :])
# 무릎 각도 및 고관절 각도 시계열 계산 (도 단위)
    vec_thigh = hip - knee
    vec_shank = ankle - knee
    cos_knee = torch.sum(vec_thigh * vec_shank, dim=-1) / (
        torch.norm(vec_thigh, dim=-1) * torch.norm(vec_shank, dim=-1) + 1e-6
    )
    theta_knee = torch.acos(torch.clamp(cos_knee, -1.0, 1.0)) * (180.0 / np.pi)
# 1. 각속도 계산 (1차 차분)
    d_knee = torch.diff(theta_knee)
# 고관절 각도 변화율 대리 변수 (대퇴골 벡터와 수직축 사이 각도)
    pitch_hip = torch.atan2(vec_thigh[:, 2], vec_thigh[:, 1]) * (180.0 / np.pi)
    d_hip = torch.diff(pitch_hip)
# Feature 1: Hip-Knee Velocity Phase Lock (HK-VPL)
    vx = d_knee - torch.mean(d_knee)
    vy = d_hip - torch.mean(d_hip)

# Feature 2: Knee-Toe Clearance (KT-DC)
